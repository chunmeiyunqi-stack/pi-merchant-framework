/**
 * Pioneer AI Service Framework 软著代码文档自动生成脚本
 * 功能：读取 copyright_source_code.txt → 生成带样式的 HTML → 自动打开浏览器 → 打印为合规 PDF
 * 使用：node generate-copyright-pdf.js
 */

const fs = require('fs');
const path = require('path');
const { execSync } = require('child_process');

// 配置项（按需修改）
const CONFIG = {
  inputFile: 'copyright_source_code.txt',      // 输入文件
  outputFile: 'PioneerAI_V1.0.0_源代码.pdf',   // 输出 PDF 文件名
  softwareName: 'Pioneer AI Service Framework',
  version: 'V1.0.0',
  copyright: '© 2026 秦晓望. All Rights Reserved.',
  submitDate: '2026 年 05 月 06 日',
  // 排版参数（严格对标软著审查规范）
  fontSize: '10.5pt',
  lineHeight: '1.15',
  pageMargin: '15mm 10mm',
  codeFont: "'Consolas', 'Courier New', monospace",
  chineseFont: "'SimSun', 'Microsoft YaHei', sans-serif"
};

// 生成 HTML 模板（含完整 CSS 样式）
function generateHTML(content) {
  return `<!DOCTYPE html>
<html lang="zh-CN">
<head>
  <meta charset="UTF-8">
  <title>${CONFIG.softwareName} ${CONFIG.version} - 源代码</title>
  <style>
    /* ========== 软著专用打印样式 ========== */
    @page {
      size: A4;
      margin: ${CONFIG.pageMargin};
      /* 页眉：严格居中 */
      @top-center {
        content: "${CONFIG.softwareName} ${CONFIG.version}";
        font-family: ${CONFIG.chineseFont};
        font-size: 9pt;
        color: #333;
        text-align: center !important;
      }
      /* 页脚：连续页码 */
      @bottom-center {
        content: "- " counter(page) " -";
        font-family: ${CONFIG.codeFont};
        font-size: 8pt;
        color: #666;
      }
    }
    
    /* 全局重置 */
    * { margin: 0; padding: 0; box-sizing: border-box; }
    
    body {
      font-family: ${CONFIG.codeFont} !important;
      font-size: ${CONFIG.fontSize} !important;
      line-height: ${CONFIG.lineHeight} !important;
      white-space: pre-wrap !important;
      word-break: break-all !important;
      color: #000;
      background: #fff;
      padding: 0;
    }
    
    /* 版权声明头部（仅首页显示） */
    .copyright-header {
      font-family: ${CONFIG.chineseFont} !important;
      font-size: ${CONFIG.fontSize} !important;
      text-align: center !important;
      margin: 0 0 20px 0 !important;
      font-weight: bold !important;
      color: #333;
      page-break-after: avoid;
    }
    
    /* 模块分隔注释 */
    .module-separator {
      color: #0066cc !important;
      font-weight: bold !important;
      margin: 15px 0 10px 0 !important;
      page-break-before: avoid;
    }
    
    /* 代码注释 */
    .comment { color: #008000 !important; font-style: italic !important; }
    
    /* 确保代码块紧凑 */
    pre, code {
      font-family: ${CONFIG.codeFont} !important;
      font-size: ${CONFIG.fontSize} !important;
      line-height: ${CONFIG.lineHeight} !important;
      white-space: pre-wrap !important;
      page-break-inside: avoid;
    }
    
    /* 打印优化 */
    @media print {
      body { -webkit-print-color-adjust: exact; print-color-adjust: exact; }
    }
  </style>
</head>
<body>
  <!-- 版权声明（仅首页） -->
  <div class="copyright-header">
    ${CONFIG.copyright}<br>
    ${CONFIG.softwareName} ${CONFIG.version} - 计算机软件著作权登记专用代码节选<br>
    提交日期：${CONFIG.submitDate}
  </div>
  
  <!-- 代码内容 -->
  <pre>${escapeHTML(content)}</pre>
  
  <!-- 自动打印脚本 -->
  <script>
    // 页面加载完成后自动触发打印
    window.addEventListener('load', () => {
      setTimeout(() => {
        if (window.location.href.includes('file://')) {
          // 本地文件：提示用户手动打印
          document.body.insertAdjacentHTML('beforeend', 
            '<div style="position:fixed;top:10px;right:10px;background:#4CAF50;color:white;padding:10px 20px;border-radius:4px;font-size:14px;z-index:9999;">🖨️ 请按 Ctrl+P 打印为 PDF</div>');
        } else {
          // 服务器环境：自动打印
          window.print();
        }
      }, 1000);
    });
  </script>
</body>
</html>`;
}

// HTML 转义函数
function escapeHTML(str) {
  return str.replace(/[&<>"']/g, m => 
    ({ '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#39;' }[m]));
}

// 主函数
function main() {
  console.log('🚀 Pioneer AI 软著 PDF 生成器启动...');
  
  // 1. 读取源代码文件
  const inputPath = path.join(__dirname, CONFIG.inputFile);
  if (!fs.existsSync(inputPath)) {
    console.error(`❌ 文件不存在: ${inputPath}`);
    console.log('💡 请先运行: node -e "..." 生成 copyright_source_code.txt');
    process.exit(1);
  }
  
  const content = fs.readFileSync(inputPath, 'utf8');
  console.log(`✅ 已读取: ${CONFIG.inputFile} (${content.length} 字符)`);
  
  // 2. 生成 HTML
  const html = generateHTML(content);
  const htmlPath = path.join(__dirname, 'copyright_temp.html');
  fs.writeFileSync(htmlPath, html, 'utf8');
  console.log(`✅ 已生成临时 HTML: ${htmlPath}`);
  
  // 3. 自动用默认浏览器打开
  const openCommand = process.platform === 'win32' 
    ? `start ${htmlPath}` 
    : process.platform === 'darwin' 
      ? `open ${htmlPath}` 
      : `xdg-open ${htmlPath}`;
  
  console.log('🌐 正在打开浏览器...');
  execSync(openCommand, { stdio: 'ignore' });
  
  // 4. 打印指引
  console.log('\n' + '='.repeat(60));
  console.log('🖨️  打印操作指引（30 秒搞定）:');
  console.log('1. 浏览器打开后，按 Ctrl+P（Mac: Cmd+P）');
  console.log('2. 目标选择: "另存为 PDF"');
  console.log('3. 纸张: A4 | 边距: 无 | 缩放: 100%');
  console.log('4. ✅ 勾选: "背景图形" + "页眉和页脚"');
  console.log('5. 保存为: ' + CONFIG.outputFile);
  console.log('='.repeat(60));
  console.log('✅ 生成成功后，请核对:');
  console.log('   • 页眉每页居中: "' + CONFIG.softwareName + ' ' + CONFIG.version + '"');
  console.log('   • 页脚页码连续: "- 1 -" 至 "- 60 -"');
  console.log('   • 首页有版权声明');
  console.log('   • 代码文字可选中（非截图）');
  console.log('\n💡 如需重新生成，直接删除 copyright_temp.html 后重跑脚本即可');
}

// 执行
main();
