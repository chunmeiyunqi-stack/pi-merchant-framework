const fs = require('fs');
const path = require('path');

// 需要合并的文件列表（按软著审查优先级排序）
const files = [
  'apps/admin/src/middleware.ts',
  'apps/web/src/middleware.ts',
  'apps/web/src/app/api/auth/pi/route.ts',
  'apps/web/src/app/api/payments/approve/route.ts',
  'apps/web/src/app/api/payments/complete/route.ts',
  'prisma/schema.prisma',
  'apps/admin/src/app/orders/page.tsx',
  'apps/admin/src/app/memberships/page.tsx',
];

// 版权声明头部
let mergedContent = `/*===========================================================
 * Pioneer AI Service Framework V1.0.0
 * 版权所有 © 2026 [秦晓望]
 * 提交日期：2026-05-06
 * 计算机软件著作权登记专用代码节选
 *===========================================================\n\n`;

let fileCount = 0;
let totalLines = 0;

files.forEach(file => {
  const filePath = path.join(__dirname, file);
  if (fs.existsSync(filePath)) {
    const content = fs.readFileSync(filePath, 'utf8');
    const lines = content.split('\n').length;
    
    mergedContent += `\n//========================================\n`;
    mergedContent += `// 模块：${file}\n`;
    mergedContent += `//========================================\n\n`;
    mergedContent += content;
    mergedContent += `\n\n`;
    
    fileCount++;
    totalLines += lines;
    console.log(`✅ 已添加: ${file} (${lines} 行)`);
  } else {
    console.log(`⚠️  文件不存在: ${file}`);
  }
});

// 输出统计信息
mergedContent += `\n/*===========================================================
 * 代码统计信息
 * 总文件数: ${fileCount}
 * 总代码行数: ${totalLines}
 * 生成时间: ${new Date().toLocaleString('zh-CN')}
 *===========================================================*/\n`;

// 写入文件
const outputPath = path.join(__dirname, 'copyright_source_code.txt');
fs.writeFileSync(outputPath, mergedContent, 'utf8');

console.log('\n' + '='.repeat(60));
console.log('✅ 合并完成！');
console.log('📁 输出文件: copyright_source_code.txt');
console.log(`📊 总文件数: ${fileCount}`);
console.log(`📊 总代码行数: ${totalLines}`);
console.log('='.repeat(60));
